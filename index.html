vehicle.userData = { speed: 0.15 + Math.random() * 0.25, type: vehicle.userData.type }; this.vehicles.push(vehicle); this.scene.add(vehicle); } } createRandomVehicle() { const rand = Math.random(); if (rand < 0.4) { return this.createCNG(); // সিএনজি } else if (rand < 0.7) { return this.createRickshaw(); // রিকশা } else { return this.createTrafficBus(); // অন্যান্য লোকাল বাস (ভিক্টর/বলাকা) } } // ১. সবুজ সিএনজি (CNG Auto Rickshaw) createCNG() { const cngGroup = new THREE.Group(); cngGroup.userData = { type: 'cng' }; const bodyGeo = new THREE.BoxGeometry(1.6, 1.6, 2.5); const bodyMat = new THREE.MeshStandardMaterial({ color: 0x15803d, roughness: 0.4 }); const body = new THREE.Mesh(bodyGeo, bodyMat); body.position.y = 0.8; body.castShadow = true; cngGroup.add(body); // সিএনজির কালো ছাদ const roofGeo = new THREE.BoxGeometry(1.62, 0.2, 2.52); const roofMat = new THREE.MeshStandardMaterial({ color: 0x0f172a }); const roof = new THREE.Mesh(roofGeo, roofMat); roof.position.y = 1.6; cngGroup.add(roof); return cngGroup; } // ২. ঢাকাইয়া রিকশা (Rickshaw) createRickshaw() { const rksGroup = new THREE.Group(); rksGroup.userData = { type: 'rickshaw' }; // রিকশার বডি ও হুড const bodyGeo = new THREE.BoxGeometry(1.3, 1.2, 2.0); const bodyMat = new THREE.MeshStandardMaterial({ color: 0xb91c1c }); // লাল হুড const body = new THREE.Mesh(bodyGeo, bodyMat); body.position.y = 0.7; body.castShadow = true; rksGroup.add(body); // রিকশার সিট const seatGeo = new THREE.BoxGeometry(1.1, 0.4, 0.8); const seatMat = new THREE.MeshStandardMaterial({ color: 0xfacc15 }); const seat = new THREE.Mesh(seatGeo, seatMat); seat.position.set(0, 0.8, -0.3); rksGroup.add(seat); return rksGroup; } // ৩. ট্রাফিক লোকাল বাস (Victor Classic / Balaka) createTrafficBus() { const busGroup = new THREE.Group(); busGroup.userData = { type: 'bus' }; const bodyGeo = new THREE.BoxGeometry(2.3, 2.5, 7.0); const busColors = [0x1d4ed8, 0xd97706, 0x7c3aed]; // নীল, কমলা, বেগুনি const chosenColor = busColors[Math.floor(Math.random() * busColors.length)]; const bodyMat = new THREE.MeshStandardMaterial({ color: chosenColor, roughness: 0.3 }); const body = new THREE.Mesh(bodyGeo, bodyMat); body.position.y = 1.3; body.castShadow = true; busGroup.add(body); return busGroup; } update(playerBusSpeed, playerBusPos) { this.vehicles.forEach(vehicle => { // বাস চললে ট্রাফিক পিছনের দিকে সরবে vehicle.position.z -= (playerBusSpeed * 10) - (vehicle.userData.speed * 3); // ট্রাফিক সামনে বা পিছনে বেশি চলে গেলে আবার রিলোড হবে if (vehicle.position.z < -20) { vehicle.position.z = 180 + Math.random() * 50; vehicle.position.x = (Math.floor(Math.random() * 3) - 1) * 3.8; } // বাসের সাথে সংঘর্ষ পরীক্ষা (Collision Detection) if (Math.abs(playerBusPos.z - vehicle.position.z) < 4.8 && Math.abs(playerBusPos.x - vehicle.position.x) < 2.0) { // হালকা ধাক্কা দেওয়া এবং ট্রাফিককে সড়িয়ে দেওয়া vehicle.position.x += (vehicle.position.x > playerBusPos.x ? 0.8 : -0.8); } }); } }
